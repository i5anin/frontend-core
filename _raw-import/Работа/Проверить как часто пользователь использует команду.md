7 дней
```sql
-- command использование /pay 
SELECT
  users.user_id,
  users.fio,
  users.role,
  COUNT(log.id) AS command_count,
  SUM(CASE WHEN DATE(log.date_time_event) = CURDATE() THEN 7 ELSE 0 END) AS week_command_count,
  MAX(log.date_time_event) AS last_command_date
FROM users
JOIN log
  ON users.user_id = log.user_id
WHERE
  log.text LIKE '/pay%' 
  AND log.type = 'command'
  AND users.fio != 'Исанин С.А.'
GROUP BY
  users.user_id,
  users.fio,
  users.role
ORDER BY
  command_count DESC;
```
